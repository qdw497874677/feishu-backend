       ```java
       /**
        * 在已初始化的话题中，将非命令纯文本合成为 /opencode chat 命令。
        * 合成后修改 message 的 content（或创建副本），使其通过正常命令路由。
        */
       private Message synthesizeCommandIfNeeded(Message message, MessageContext messageContext) {
           String content = message.getContent();
           if (content == null || content.trim().isEmpty()) {
               return message;
           }
           String trimmed = content.trim();
           
           // 如果已经是命令（以 / 开头），不合成
           if (trimmed.startsWith("/")) {
               return message;
           }
           
           // 必须在已初始化话题中
           if (!messageContext.isResolved() || !messageContext.isThreadContext()) {
               return message;
           }
           
           TopicState state = sessionManager.detectTopicState(messageContext);
           if (state != TopicState.INITIALIZED) {
               // 未初始化话题中的纯文本：不合成，但稍后在 handleMessageInternal 中会显示引导
               return message;
           }
           
           // 合成为 /opencode chat <原文>
           log.info("合成纯文本为 chat 命令: topicId={}, content='{}'", 
                    message.getTopicId(), trimmed.substring(0, Math.min(50, trimmed.length())));
           
           Message synthesized = message.withContent("/opencode chat " + trimmed);
           return synthesized;
       }
       ```

       ```java
       @Component
       public class NextStepSuggester {
       
           /**
            * 根据刚执行的命令和当前状态，生成下一步建议。
            * 
            * @param executedCommand 刚执行的子命令（如 "projects", "sc" 等）
            * @param state 当前话题状态
            * @param messageContext 消息上下文（可选，用于获取绑定信息）
            * @return 建议文本，或 null 表示不需要建议
            */
           public String suggest(String executedCommand, TopicState state, MessageContext messageContext) {
               return switch (executedCommand) {
                   case "projects", "p" -> "💡 下一步：`/oc sessions <项目名>` 查看会话列表";
                   case "sessions", "s" -> "💡 下一步：`/oc sc <会话ID>` 绑定会话到当前话题";
                   case "sc", "session" -> suggestAfterSessionBind(state);
                   case "connect" -> "💡 下一步：`/oc projects` 查看项目列表";
                   case "reset" -> "💡 下一步：`/oc sc <会话ID>` 重新绑定，或 `/oc sessions` 查看会话";
                   case "status" -> suggestAfterStatus(state);
                   case "new" -> "💡 下一步：直接输入问题开始对话";
                   case "chat", "chatnow", "cn" -> null; // 已在对话中，不需要提示
                   case "commands", "help" -> null; // 信息类命令不需要提示
                   default -> null;
               };
           }
           
           private String suggestAfterSessionBind(TopicState state) {
               if (state == TopicState.INITIALIZED) {
                   return "💡 下一步：直接输入问题开始对话，或 `/oc chat <内容>`";
               }
               return "💡 下一步：`/oc chat <内容>` 开始对话";
           }
           
           private String suggestAfterStatus(TopicState state) {
               if (state == TopicState.INITIALIZED) {
                   return "💡 下一步：直接输入问题继续对话";
               }
               return "💡 下一步：`/oc sc <会话ID>` 绑定会话";
           }
       }
       ```

    2. **新建 `prependStatusIndicator()` 私有方法**：
       ```java
       /**
        * 为 OpenCode 应用的回复添加状态行。
        * 格式：📎 {项目名或上下文} | {会话ID前12位}
        * 
        * 排除：非 OpenCode 应用、null/空回复、help 命令
        */
       private String prependStatusIndicator(String replyContent, FishuAppI app, 
                                              MessageContext messageContext, Message message) {
           // 仅 OpenCode 应用
           if (!"opencode".equals(app.getAppId())) {
               return replyContent;
           }
           // null 或空内容不添加（noReply 路径）
           if (replyContent == null || replyContent.trim().isEmpty()) {
               return replyContent;
           }
           // help 命令跳过（帮助文本已足够长）
           if (isHelpCommand(message)) {
               return replyContent;
           }
           
           String statusLine = buildStatusLine(messageContext);
           if (statusLine == null) {
               return replyContent;
           }
           return statusLine + "\n\n" + replyContent;
       }
       
       private boolean isHelpCommand(Message message) {
           String content = message.getContent();
           if (content == null) return false;
           String trimmed = content.trim().toLowerCase();
           // 匹配 /opencode help 或 /oc help
           return trimmed.matches("/(?:opencode|oc|code)\\s+help.*");
       }
       
       private String buildStatusLine(MessageContext messageContext) {
           if (messageContext == null || !messageContext.isResolved()) {
               return null;
           }
           if (!messageContext.isBoundToApp("opencode")) {
               return null; // 无 OpenCode 绑定，不显示
           }
           
           ImContextBinding binding = messageContext.getBinding();
           String sessionId = binding.getSessionId();
           
           if (sessionId != null) {
               // 有会话：显示状态
               // sessionId 是内部 UUID，需要获取 OpenCode ses_xxx ID
               // 使用 SessionManager 获取 OpenCode session ID
               String displayId = getDisplaySessionId(messageContext);
               return "📎 opencode | " + (displayId != null ? displayId : sessionId.substring(0, Math.min(12, sessionId.length())));
           } else {
               return "📎 opencode | 未绑定会话";
           }
       }
       ```
